BFS (Breadth First Search) - level by level traversal using a queue
--------------------------------------------------------------------
Time: O(V + E)   Space: O(V)

Standard code (adj is an adjacency list):

    void bfs(vector<vector<int>>& adj, vector<int>& res, vector<int>& visited, int src) {
        queue<int> q;
        q.push(src);
        visited[src] = 1;
        while (!q.empty()) {
            int node = q.front();
            q.pop();
            res.push_back(node);
            for (int nb : adj[node]) {
                if (!visited[nb]) {
                    visited[nb] = 1;     // mark when PUSHING
                    q.push(nb);
                }
            }
        }
    }

    vector<int> bfsTraversal(int n, vector<vector<int>>& adj) {
        vector<int> visited(n, 0), res;
        bfs(adj, res, visited, 0);     // only nodes reachable from 0
        // to cover disconnected graphs, instead loop:
        // for (int i = 0; i < n; i++) if (!visited[i]) bfs(adj, res, visited, i);
        return res;
    }


Common mistakes (I made all of these):
---------------------------------------
1. Forgetting q.push(nb)  -> only the start node and its direct neighbours get visited.
2. Marking visited when POPPING instead of when pushing
    -> the same node gets pushed many times and appears twice in the result.
3. Treating an adjacency list as an edge list (see 01-GraphRepresentation.txt).
4. Looping over all nodes when the problem says "start from node 0"
    -> extra nodes from other components get added.

Multi-source BFS (Rotten Oranges, 01 Matrix, Number of Enclaves):
------------------------------------------------------------------
Push ALL sources into the queue at the start (e.g. all rotten oranges), then run normal BFS.
Every source spreads at the same time, so the level = time taken.
