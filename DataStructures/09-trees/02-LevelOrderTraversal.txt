Level order traversal (BFS on a tree)
-------------------------------------
Base for: left/right view, zigzag traversal, top/bottom view, vertical order, max width.

vector<vector<int>> levelOrder(TreeNode* root) {
    vector<vector<int>> res;
    if (root == NULL) return res;
    queue<TreeNode*> q;
    q.push(root);
    while (!q.empty()) {
        int size = q.size();          // number of nodes on this level
        vector<int> level;
        for (int i = 0; i < size; i++) {
            TreeNode* node = q.front();
            q.pop();
            level.push_back(node->val);
            if (node->left) q.push(node->left);
            if (node->right) q.push(node->right);
        }
        res.push_back(level);
    }
    return res;
}

Zigzag: same code, reverse `level` on every odd level (or fill it from the back).
