# Lesson 5: Bulk CRM sync implementation

The code below is vendor-neutral. Adapt the URL and response fields for Zendesk, Salesforce, HubSpot, or another CRM.

## 1. Page model and async client

```python
# app/integrations/crm/client.py
import asyncio
from dataclasses import dataclass
from typing import Optional

import requests


@dataclass
class ContactPage:
    items: list[dict]
    next_cursor: Optional[str]
    has_more: bool


class CRMClient:
    def __init__(self, base_url: str, token: str):
        self.base_url = base_url
        self.token = token

    def _fetch_contacts_sync(
        self,
        cursor: Optional[str],
        page_size: int,
    ) -> ContactPage:
        response = requests.get(
            f"{self.base_url}/contacts",
            headers={"Authorization": f"Bearer {self.token}"},
            params={"cursor": cursor, "limit": page_size},
            timeout=30,
        )
        response.raise_for_status()
        data = response.json()
        return ContactPage(
            items=data.get("items", []),
            next_cursor=data.get("next_cursor"),
            has_more=bool(data.get("has_more", False)),
        )

    async def fetch_contacts(
        self,
        cursor: Optional[str],
        page_size: int = 500,
    ) -> ContactPage:
        return await asyncio.to_thread(
            self._fetch_contacts_sync,
            cursor,
            page_size,
        )
```

Why this split?

`requests` is synchronous. The private sync method contains the blocking work; the public async method moves it to a thread.

Never log the bearer token.

## 2. Converter

```python
# app/integrations/crm/converter.py
from datetime import datetime, timezone


def convert_contact(tenant: str, source: dict) -> dict:
    return {
        "tenant": tenant,
        "external_id": str(source["id"]),
        "name": source.get("name") or "",
        "email": source.get("email"),
        "external_updated_at": source.get("updated_at"),
        "synced_at": datetime.now(timezone.utc),
        "is_deleted": bool(source.get("deleted", False)),
    }
```

Why use a converter?

- external field names do not spread through your application;
- defaults and type conversions live in one place;
- changing CRM vendors does not require rewriting database code.

Decide how malformed records behave. For a large sync, recording and skipping a bad record is often better than losing every valid record on the page.

## 3. Repository with bulk upsert

```python
# app/integrations/crm/repository.py
from datetime import datetime, timezone
from typing import Optional

from pymongo import UpdateOne


class CRMRepository:
    def __init__(self, contacts, checkpoints):
        self.contacts = contacts
        self.checkpoints = checkpoints

    def upsert_contacts(self, tenant: str, documents: list[dict]):
        if not documents:
            return 0

        operations = [
            UpdateOne(
                {
                    "tenant": tenant,
                    "external_id": document["external_id"],
                },
                {"$set": document},
                upsert=True,
            )
            for document in documents
        ]

        result = self.contacts.bulk_write(operations, ordered=False)
        return result.upserted_count + result.modified_count

    def load_cursor(self, tenant: str) -> Optional[str]:
        checkpoint = self.checkpoints.find_one(
            {"tenant": tenant, "job": "crm_contacts"}
        )
        return checkpoint.get("cursor") if checkpoint else None

    def save_cursor(self, tenant: str, cursor: Optional[str]):
        self.checkpoints.update_one(
            {"tenant": tenant, "job": "crm_contacts"},
            {
                "$set": {
                    "cursor": cursor,
                    "updated_at": datetime.now(timezone.utc),
                }
            },
            upsert=True,
        )
```

`ordered=False` allows MongoDB to continue independent operations when one operation fails. Inspect and log bulk-write failures according to your data policy.

## 4. Synchronize one tenant

```python
# app/jobs/sync_crm.py
from app.common.logger import logger
from app.integrations.crm.converter import convert_contact


async def sync_tenant(
    tenant: str,
    client,
    repository,
    full_sync: bool = False,
):
    cursor = None if full_sync else repository.load_cursor(tenant)
    pages = 0
    received = 0
    written = 0

    while True:
        page = await client.fetch_contacts(
            cursor=cursor,
            page_size=500,
        )

        documents = []
        for item in page.items:
            try:
                documents.append(convert_contact(tenant, item))
            except Exception as exc:
                logger.warning(
                    "Contact skipped | tenant=%s | external_id=%s | error=%s",
                    tenant,
                    item.get("id"),
                    exc,
                )

        page_written = repository.upsert_contacts(tenant, documents)

        # Save only after the database write succeeds.
        repository.save_cursor(tenant, page.next_cursor)

        pages += 1
        received += len(page.items)
        written += page_written

        logger.info(
            "CRM page complete | tenant=%s | pages=%d | received=%d | written=%d",
            tenant,
            pages,
            received,
            written,
        )

        if not page.has_more:
            break

        if not page.next_cursor:
            raise RuntimeError("CRM returned has_more without next_cursor")

        cursor = page.next_cursor

    return {
        "tenant": tenant,
        "pages": pages,
        "received": received,
        "written": written,
    }
```

This function never holds all contacts in memory. A failed database write occurs before checkpoint advancement, so the same page can be retried.

## 5. Orchestrate tenants

```python
async def run(tenants: list[str], full_sync: bool = False):
    results = []

    for tenant in tenants:
        try:
            client = await build_client_for_tenant(tenant)
            repository = await build_repository_for_tenant(tenant)
            result = await sync_tenant(
                tenant,
                client,
                repository,
                full_sync=full_sync,
            )
            results.append(result)
        except Exception as exc:
            logger.exception(
                "CRM tenant sync failed | tenant=%s | error=%s",
                tenant,
                exc,
            )

    return results
```

One tenant failure does not prevent later tenants from running. Sequential execution protects shared resources and respects external API limits.

## 6. Command-line entry point

```python
import argparse
import asyncio


def parse_args():
    parser = argparse.ArgumentParser()
    parser.add_argument("--tenant", action="append", dest="tenants")
    parser.add_argument("--full", action="store_true")
    return parser.parse_args()


if __name__ == "__main__":
    args = parse_args()
    selected_tenants = args.tenants or load_enabled_tenants()
    asyncio.run(run(selected_tenants, full_sync=args.full))
```

Example commands:

```bash
# Incremental sync for one tenant
python -m app.jobs.sync_crm --tenant acme

# Incremental sync for selected tenants
python -m app.jobs.sync_crm --tenant acme --tenant globex

# Full sync for one tenant
python -m app.jobs.sync_crm --tenant acme --full
```

`python -m` executes the module using the package import path, so imports such as `from app...` resolve consistently when run from the repository root.

## 7. Safe concurrency later

If sequential execution is too slow, use a small semaphore:

```python
semaphore = asyncio.Semaphore(3)


async def sync_with_limit(tenant: str):
    async with semaphore:
        return await sync_one_configured_tenant(tenant)


results = await asyncio.gather(
    *(sync_with_limit(tenant) for tenant in tenants),
    return_exceptions=True,
)
```

The limit of three is an example, not a universal value. Choose it from API rate limits, database capacity, and measurements.
