# Lesson 3: Scheduled email implementation

This implementation uses MongoDB for durable job records and a separate polling worker.

## 1. Request and response models

```python
# app/emails/models/email.py
from datetime import datetime, timezone

from pydantic import BaseModel, Field, field_validator


class ScheduleEmailRequest(BaseModel):
    to: str = Field(min_length=3, max_length=320)
    subject: str = Field(min_length=1, max_length=200)
    body: str = Field(min_length=1, max_length=10_000)
    run_at: datetime

    @field_validator("run_at")
    @classmethod
    def normalize_run_at(cls, value: datetime) -> datetime:
        if value.tzinfo is None:
            raise ValueError("run_at must include a timezone")
        value = value.astimezone(timezone.utc)
        if value <= datetime.now(timezone.utc):
            raise ValueError("run_at must be in the future")
        return value


class ScheduleEmailResponse(BaseModel):
    job_id: str
    status: str
    run_at: datetime
```

The validation rejects ambiguous local time and past schedules before creating a job.

## 2. Job service

```python
# app/emails/services/email_jobs.py
from datetime import datetime, timedelta, timezone
from uuid import uuid4

from pymongo import ReturnDocument

from app.emails.models.email import (
    ScheduleEmailRequest,
    ScheduleEmailResponse,
)


def utc_now() -> datetime:
    return datetime.now(timezone.utc)


def schedule_email(collection, payload: ScheduleEmailRequest):
    job_id = str(uuid4())
    document = {
        "job_id": job_id,
        "type": "send_email",
        "status": "pending",
        "run_at": payload.run_at,
        "payload": {
            "to": payload.to,
            "subject": payload.subject,
            "body": payload.body,
        },
        "attempts": 0,
        "max_attempts": 3,
        "locked_at": None,
        "last_error": None,
        "created_at": utc_now(),
    }
    collection.insert_one(document)
    return ScheduleEmailResponse(
        job_id=job_id,
        status="pending",
        run_at=payload.run_at,
    )


def claim_due_email(collection):
    now = utc_now()
    return collection.find_one_and_update(
        {
            "type": "send_email",
            "status": "pending",
            "run_at": {"$lte": now},
            "$expr": {"$lt": ["$attempts", "$max_attempts"]},
        },
        {
            "$set": {
                "status": "processing",
                "locked_at": now,
            },
            "$inc": {"attempts": 1},
        },
        sort=[("run_at", 1)],
        return_document=ReturnDocument.AFTER,
    )


def mark_succeeded(collection, job_id: str):
    collection.update_one(
        {"job_id": job_id, "status": "processing"},
        {
            "$set": {
                "status": "succeeded",
                "completed_at": utc_now(),
                "last_error": None,
            }
        },
    )


def mark_failed_or_retry(collection, job: dict, error: Exception):
    exhausted = job["attempts"] >= job["max_attempts"]
    next_status = "failed" if exhausted else "pending"
    retry_delay = timedelta(seconds=30 * job["attempts"])

    collection.update_one(
        {"job_id": job["job_id"], "status": "processing"},
        {
            "$set": {
                "status": next_status,
                "run_at": utc_now() + retry_delay,
                "locked_at": None,
                "last_error": str(error)[:1000],
            }
        },
    )
```

The retry delay grows with the attempt count. Errors are truncated so one exception cannot create an unbounded document.

## 3. API endpoint

```python
# app/emails/api/v1/email.py
from fastapi import APIRouter

from app.emails.models.email import (
    ScheduleEmailRequest,
    ScheduleEmailResponse,
)
from app.emails.services.email_jobs import schedule_email
from app.db import jobs_collection


router = APIRouter()


@router.post("/schedule", response_model=ScheduleEmailResponse)
async def api_schedule_email(payload: ScheduleEmailRequest):
    return schedule_email(jobs_collection, payload)
```

The endpoint stores the job; it does not sleep until `run_at` and does not send the email itself.

## 4. Email provider wrapper

```python
# app/emails/services/email_client.py
import asyncio


def _send_email_sync(job: dict):
    payload = job["payload"]
    email_provider.send(
        to=payload["to"],
        subject=payload["subject"],
        body=payload["body"],
        idempotency_key=job["job_id"],
        timeout=10,
    )


async def send_email(job: dict):
    await asyncio.to_thread(_send_email_sync, job)
```

The provider SDK is synchronous, so `asyncio.to_thread` prevents it from blocking the worker's event loop.

## 5. Separate worker

```python
# app/jobs/email_worker.py
import asyncio

from app.common.logger import logger
from app.db import jobs_collection
from app.emails.services.email_client import send_email
from app.emails.services.email_jobs import (
    claim_due_email,
    mark_failed_or_retry,
    mark_succeeded,
)


POLL_SECONDS = 2


async def run_worker():
    logger.info("Email worker started")

    while True:
        job = claim_due_email(jobs_collection)

        if not job:
            await asyncio.sleep(POLL_SECONDS)
            continue

        try:
            await send_email(job)
            mark_succeeded(jobs_collection, job["job_id"])
            logger.info("Email job succeeded | job_id=%s", job["job_id"])
        except Exception as exc:
            logger.exception(
                "Email job failed | job_id=%s | error=%s",
                job["job_id"],
                exc,
            )
            mark_failed_or_retry(jobs_collection, job, exc)


if __name__ == "__main__":
    asyncio.run(run_worker())
```

Run the API and worker as different processes:

```bash
uvicorn app.main:app --host 0.0.0.0 --port 8000
python -m app.jobs.email_worker
```

## 6. Required MongoDB indexes

```python
jobs_collection.create_index(
    [("status", 1), ("type", 1), ("run_at", 1)]
)
jobs_collection.create_index("job_id", unique=True)
```

The first index supports due-job polling. The unique index prevents duplicate job IDs.

## 7. Recover abandoned jobs

A worker can crash after claiming. Periodically return old processing jobs to pending:

```python
stale_before = utc_now() - timedelta(minutes=5)

jobs_collection.update_many(
    {
        "status": "processing",
        "locked_at": {"$lt": stale_before},
    },
    {
        "$set": {
            "status": "pending",
            "locked_at": None,
        }
    },
)
```

Choose the stale timeout longer than the normal maximum email-send duration.
