# Module 2 reference implementation

This compact example uses the same patterns as `customer-success`: models, async services, nested routers, and `HTTPException`.

## `app/tasks/models/task.py`

```python
from typing import Optional

from pydantic import BaseModel, Field


class TaskCreateRequest(BaseModel):
    title: str = Field(min_length=2, max_length=100)
    description: Optional[str] = None


class TaskCreateResponse(BaseModel):
    success: bool
    task_id: str
    msg: str


class TaskListItem(BaseModel):
    task_id: str
    title: str
    description: Optional[str] = None


class TaskListResponse(BaseModel):
    items: list[TaskListItem]
    has_more: bool
```

## `app/tasks/services/task.py`

```python
from uuid import uuid4

from fastapi import HTTPException

from app.tasks.models.task import (
    TaskCreateRequest,
    TaskCreateResponse,
    TaskListItem,
    TaskListResponse,
)


tasks: dict[str, dict] = {}


async def create_task(
    tenant: str,
    payload: TaskCreateRequest,
) -> TaskCreateResponse:
    task_id = str(uuid4())
    tasks[task_id] = {
        "task_id": task_id,
        "tenant": tenant,
        "title": payload.title,
        "description": payload.description,
    }
    return TaskCreateResponse(
        success=True,
        task_id=task_id,
        msg="Task successfully created",
    )


async def list_tasks(
    tenant: str,
    limit: int,
    offset: int,
) -> TaskListResponse:
    tenant_tasks = [
        TaskListItem(**task)
        for task in tasks.values()
        if task["tenant"] == tenant
    ]
    page = tenant_tasks[offset : offset + limit + 1]
    return TaskListResponse(
        items=page[:limit],
        has_more=len(page) > limit,
    )


async def get_task(tenant: str, task_id: str) -> TaskListItem:
    task = tasks.get(task_id)
    if not task or task["tenant"] != tenant:
        raise HTTPException(status_code=404, detail="Task not found")
    return TaskListItem(**task)


async def delete_task(tenant: str, task_id: str):
    await get_task(tenant, task_id)
    del tasks[task_id]
    return {"success": True, "msg": "Task successfully deleted"}
```

Pydantic ignores the internal `tenant` field when building `TaskListItem`, so it is not returned.

## `app/tasks/api/v1/task.py`

```python
from fastapi import APIRouter, HTTPException, Query

from app.common.logger import logger
from app.tasks.models.task import (
    TaskCreateRequest,
    TaskCreateResponse,
    TaskListItem,
    TaskListResponse,
)
from app.tasks.services.task import (
    create_task,
    delete_task,
    get_task,
    list_tasks,
)


router = APIRouter()


@router.post("", response_model=TaskCreateResponse)
async def api_create_task(tenant: str, payload: TaskCreateRequest):
    try:
        return await create_task(tenant, payload)
    except HTTPException:
        raise
    except Exception as exc:
        logger.exception("[%s] Unexpected create error: %s", tenant, exc)
        raise HTTPException(status_code=500, detail="Internal Server Error")


@router.get("", response_model=TaskListResponse)
async def api_list_tasks(
    tenant: str,
    limit: int = Query(20, ge=1, le=100),
    offset: int = Query(0, ge=0),
):
    return await list_tasks(tenant=tenant, limit=limit, offset=offset)


@router.get("/{task_id}", response_model=TaskListItem)
async def api_get_task(tenant: str, task_id: str):
    try:
        return await get_task(tenant, task_id)
    except HTTPException:
        raise
    except Exception as exc:
        logger.exception("[%s] Unexpected get error: %s", tenant, exc)
        raise HTTPException(status_code=500, detail="Internal Server Error")


@router.delete("/{task_id}")
async def api_delete_task(tenant: str, task_id: str):
    return await delete_task(tenant, task_id)
```

## `app/tasks/api/v1/router.py`

```python
from fastapi import APIRouter

from app.tasks.api.v1 import task


router = APIRouter(prefix="/api/v1", tags=["Tasks"])
router.include_router(task.router, prefix="/{tenant}/tasks")
```

## Additions to `app/main.py`

```python
from fastapi import FastAPI
from fastapi.exceptions import RequestValidationError

from app.common.validation_errors import validation_exception_handler
from app.tasks.api.v1.router import router as tasks_v1_router


app = FastAPI(docs_url="/swagger")
app.add_exception_handler(
    RequestValidationError,
    validation_exception_handler,
)
app.include_router(tasks_v1_router)
```
