# Module 4 reference implementation

```python
import logging
from typing import Literal, Optional

from pydantic import BaseModel


logger = logging.getLogger(__name__)

SUPPORTED_INTENTS = {
    "get_account",
    "list_open_tickets",
    "get_account_health",
}


class ScopeClassification(BaseModel):
    outcome: Literal[
        "supported",
        "needs_clarification",
        "out_of_scope",
    ]
    intent: Optional[Literal[
        "get_account",
        "list_open_tickets",
        "get_account_health",
    ]] = None
    account_id: Optional[str] = None


async def classify_scope(question, client, model_name):
    try:
        response = await client.responses.parse(
            model=model_name,
            instructions=SCOPE_INSTRUCTIONS,
            input=question,
            text_format=ScopeClassification,
        )
    except Exception:
        logger.exception("Scope classification failed")
        return None

    return response.output_parsed


async def apply_scope_gate(question, context, services):
    if not question.strip() or len(question) > 2000:
        return out_of_scope_response()

    classification = await classify_scope(
        question,
        services.openai_client,
        services.classifier_model,
    )

    if classification is None:
        return temporarily_unavailable_response()

    if classification.outcome == "out_of_scope":
        return out_of_scope_response()

    if classification.outcome == "needs_clarification":
        return clarification_response("Which account do you mean?")

    if classification.intent not in SUPPORTED_INTENTS:
        return out_of_scope_response()

    if not classification.account_id:
        return clarification_response("Which account do you mean?")

    return await services.tools.answer(
        question=question,
        context=context,
    )
```

The classifier's `account_id` helps validate completeness, but the Module 3 tool call still creates and validates the actual execution arguments. Module 2 remains responsible for authorization.
