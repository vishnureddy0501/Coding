```python
import json
import time


def load_cases(path):
    with open(path, encoding="utf-8") as file:
        return [json.loads(line) for line in file if line.strip()]


async def evaluate_case(case, orchestrator, context_factory):
    context = context_factory(case["user"])
    started_at = time.perf_counter()
    response = await orchestrator.answer_question(case["question"], context)
    latency_ms = round((time.perf_counter() - started_at) * 1000, 2)

    status_correct = response.status == case["expected_status"]
    expected_tool = case.get("expected_tool")
    actual_tools = [source.name for source in response.sources]
    tool_correct = (
        expected_tool is None or expected_tool in actual_tools
    )

    return {
        "id": case["id"],
        "status_correct": status_correct,
        "tool_correct": tool_correct,
        "actual_status": response.status,
        "actual_tools": actual_tools,
        "latency_ms": latency_ms,
    }
```

This checks deterministic properties. For answer correctness, compare required facts with the captured tool result or review the case manually.

Never use real production secrets or sensitive customer data in the evaluation dataset.
