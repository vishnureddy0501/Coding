```python
import asyncio
import logging


logger = logging.getLogger(__name__)


class ChatOrchestrator:
    def __init__(self, scope, tools, timeout_seconds=15):
        self.scope = scope
        self.tools = tools
        self.timeout_seconds = timeout_seconds

    async def answer_question(self, question, context):
        try:
            return await asyncio.wait_for(
                self._answer(question, context),
                timeout=self.timeout_seconds,
            )
        except TimeoutError:
            logger.warning("Chat request timed out")
            return temporarily_unavailable_response()
        except Exception:
            logger.exception("Unexpected chat failure")
            return temporarily_unavailable_response()

    async def _answer(self, question, context):
        classification = await self.scope.classify(question)

        if classification is None:
            return temporarily_unavailable_response()

        if classification.outcome == "out_of_scope":
            return out_of_scope_response()

        if classification.outcome == "needs_clarification":
            return clarification_response("Which account do you mean?")

        return await self.tools.answer(question, context)
```

```python
router = APIRouter(prefix="/api/v1", tags=["Chat"])


@router.post("/chat", response_model=ChatResponse)
async def chat(payload: ChatRequest, request: Request):
    context = build_request_context(request)
    return await request.app.state.chat_orchestrator.answer_question(
        payload.question,
        context,
    )


@asynccontextmanager
async def lifespan(app: FastAPI):
    app.state.chat_orchestrator = create_chat_orchestrator()
    yield
    await close_shared_clients(app.state.chat_orchestrator)


app = FastAPI(lifespan=lifespan)
app.include_router(router)
```

Shared database, HTTP, and model clients are created once during startup and closed during shutdown.
