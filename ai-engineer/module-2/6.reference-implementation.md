# Module 2 reference implementation

This is the smallest useful package shape for the three approved operations.

```text
app/chat_data/
├── context.py
├── models.py
├── repository.py
├── api_client.py
└── service.py
```

## `context.py`

```python
from typing import Literal

from pydantic import BaseModel, Field


class RequestContext(BaseModel):
    tenant: str = Field(min_length=1)
    user_id: str = Field(min_length=1)
    role: Literal["account_manager", "support_agent", "tenant_admin"]
    allowed_account_ids: list[str] = Field(default_factory=list)
```

## `models.py`

```python
from datetime import datetime
from typing import Generic, Literal, Optional, TypeVar

from pydantic import BaseModel, Field


class AccountData(BaseModel):
    account_id: str
    name: str
    status: Literal["active", "at_risk", "inactive"]
    owner_name: Optional[str] = None
    last_updated_at: datetime


class TicketData(BaseModel):
    ticket_id: str
    subject: str
    priority: Literal["low", "medium", "high", "urgent"]
    created_at: datetime


class TicketListData(BaseModel):
    items: list[TicketData]
    has_more: bool


class AccountHealthData(BaseModel):
    account_id: str
    health: Literal["healthy", "watch", "at_risk"]
    score: int = Field(ge=0, le=100)
    updated_at: datetime


class DataSource(BaseModel):
    operation: str
    entity_ids: list[str]
    retrieved_at: datetime
    truncated: bool = False


T = TypeVar("T")


class ReadResult(BaseModel, Generic[T]):
    status: Literal["found", "not_found", "unavailable"]
    data: Optional[T] = None
    source: Optional[DataSource] = None
```

## `repository.py`

```python
from datetime import datetime, timezone

from app.chat_data.context import RequestContext
from app.chat_data.models import AccountData, DataSource, TicketData, TicketListData


MAX_TICKETS = 50


async def read_account(collection, context: RequestContext, account_id: str):
    if context.role != "tenant_admin" and account_id not in context.allowed_account_ids:
        return None

    document = await collection.find_one(
        {
            "tenant": context.tenant,
            "account_id": account_id,
            "is_deleted": False,
        },
        {
            "_id": 0,
            "account_id": 1,
            "name": 1,
            "status": 1,
            "owner_name": 1,
            "last_updated_at": 1,
        },
    )
    if document is None:
        return None

    return (
        AccountData.model_validate(document),
        DataSource(
            operation="get_account",
            entity_ids=[document["account_id"]],
            retrieved_at=datetime.now(timezone.utc),
        ),
    )


async def read_open_tickets(
    collection,
    context: RequestContext,
    account_id: str,
    limit: int,
):
    if context.role != "tenant_admin" and account_id not in context.allowed_account_ids:
        return None

    safe_limit = min(max(limit, 1), MAX_TICKETS)
    cursor = collection.find(
        {
            "tenant": context.tenant,
            "account_id": account_id,
            "status": {"$in": ["open", "pending"]},
            "is_deleted": False,
        },
        {
            "_id": 0,
            "ticket_id": 1,
            "subject": 1,
            "priority": 1,
            "created_at": 1,
        },
    ).sort("created_at", -1).limit(safe_limit + 1)

    documents = await cursor.to_list(length=safe_limit + 1)
    has_more = len(documents) > safe_limit
    data = TicketListData(
        items=[TicketData.model_validate(item) for item in documents[:safe_limit]],
        has_more=has_more,
    )
    source = DataSource(
        operation="list_open_tickets",
        entity_ids=[item.ticket_id for item in data.items],
        retrieved_at=datetime.now(timezone.utc),
        truncated=has_more,
    )
    return data, source
```

## `api_client.py`

```python
import httpx


class AccountHealthClient:
    def __init__(self, base_url: str, service_token: str):
        self.base_url = base_url.rstrip("/")
        self.service_token = service_token

    async def get_health(self, tenant: str, account_id: str) -> dict:
        async with httpx.AsyncClient(timeout=5.0) as client:
            response = await client.get(
                f"{self.base_url}/internal/accounts/{account_id}/health",
                headers={
                    "Authorization": f"Bearer {self.service_token}",
                    "X-Tenant": tenant,
                },
            )
            response.raise_for_status()
            return response.json()
```

## `service.py`

```python
import logging
from datetime import datetime, timezone

import httpx

from app.chat_data.context import RequestContext
from app.chat_data.models import AccountHealthData, DataSource, ReadResult
from app.chat_data.repository import read_account, read_open_tickets


logger = logging.getLogger(__name__)


class ChatDataService:
    def __init__(self, accounts, tickets, health_client):
        self.accounts = accounts
        self.tickets = tickets
        self.health_client = health_client

    async def get_account(self, context: RequestContext, account_id: str):
        try:
            result = await read_account(self.accounts, context, account_id)
        except Exception:
            logger.exception("Account data source failed")
            return ReadResult(status="unavailable")

        if result is None:
            return ReadResult(status="not_found")

        data, source = result
        return ReadResult(status="found", data=data, source=source)

    async def list_open_tickets(
        self,
        context: RequestContext,
        account_id: str,
        limit: int = 20,
    ):
        try:
            result = await read_open_tickets(
                self.tickets,
                context,
                account_id,
                limit,
            )
        except Exception:
            logger.exception("Ticket data source failed")
            return ReadResult(status="unavailable")

        if result is None:
            return ReadResult(status="not_found")

        data, source = result
        return ReadResult(status="found", data=data, source=source)

    async def get_account_health(
        self,
        context: RequestContext,
        account_id: str,
    ):
        if context.role != "tenant_admin" and account_id not in context.allowed_account_ids:
            return ReadResult(status="not_found")

        try:
            payload = await self.health_client.get_health(
                tenant=context.tenant,
                account_id=account_id,
            )
            data = AccountHealthData.model_validate(payload)
        except httpx.HTTPStatusError as exc:
            if exc.response.status_code == 404:
                return ReadResult(status="not_found")
            logger.exception("Account health API rejected the request")
            return ReadResult(status="unavailable")
        except Exception:
            logger.exception("Account health source failed")
            return ReadResult(status="unavailable")

        return ReadResult(
            status="found",
            data=data,
            source=DataSource(
                operation="get_account_health",
                entity_ids=[data.account_id],
                retrieved_at=datetime.now(timezone.utc),
            ),
        )
```

## Why this structure is enough

- `context.py` carries trusted identity and permission scope.
- `models.py` defines exactly what later chatbot code may receive.
- `repository.py` owns fixed database queries and projections.
- `api_client.py` owns the fixed internal HTTP call and credentials.
- `service.py` exposes only approved operations and normalizes failures.

Module 3 can now expose these three service methods as model tools without exposing collections, filters, URLs, or credentials.
